import pandas as pd

def calculate_special_bonus(ep: pd.DataFrame) -> pd.DataFrame:
    ep['bonus'] = 0
    ep.loc[(ep['employee_id'] % 2 != 0) & (~ep['name'].str.startswith('M')), 'bonus'] = ep['salary']
    ans = ep[['employee_id', 'bonus']].sort_values(by='employee_id', ascending=True)
    
    return ans
